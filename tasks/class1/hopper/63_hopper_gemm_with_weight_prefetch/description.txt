      __nv_fp8_e4m3 const *A, int lda,

/*
 * Hopper GEMM with Weight Prefetch test harness — FP8 precision
 * Based on CUTLASS example 63_hopper_gemm_with_weight_prefetch
 *
 * The LLM must implement:
 *   cudaError_t HopperGemmPrefetch(int M, int N, int K, float alpha,
 *       __nv_fp8_e4m3 const *A, int lda,
 *       __nv_fp8_e5m2 const *B, int ldb,
 *       float beta,
 *       __nv_fp8_e4m3 *D, int ldd);
 *
 * A is row-major [M, K] (lda = K) in FP8 E4M3,
 * B is column-major [K, N] (ldb = K) in FP8 E5M2,
 * D is row-major [M, N] (ldd = N) in FP8 E4M3.
 * FP32 accumulation internally.
 *
 * Compile: nvcc -O3 harness.cu solution.cu -o test && ./test
 */

#include <cuda_fp8.h>
#include <iostream>
#include <vector>
#include <cmath>
#include "../../include/kh_benchmark.h"

#define CUDA_CHECK(status)                                              \
  {                                                                     \
    cudaError_t error = status;                                         \
    if (error != cudaSuccess) {                                         \
      std::cerr << "Got bad cuda status: " << cudaGetErrorString(error) \
                << " at line: " << __LINE__ << std::endl;               \
      exit(EXIT_FAILURE);                                               \
    }                                                                   \
  }

extern cudaError_t HopperGemmPrefetch(
  int M, int N, int K, float alpha,
  __nv_fp8_e4m3 const *A, int lda,
  __nv_fp8_e5m2 const *B, int ldb,
  float beta,
  __nv_fp8_e4m3 *D, int ldd);

/// CPU reference: A row-major, B col-major, D row-major (all in float)
void reference(int M, int N, int K, float alpha,
  std::vector<float> const &A, int lda,
  std::vector<float> const &B, int ldb,
  float beta,
  std::vector<float> &D, int ldd) {
  for (int i = 0; i < M; ++i)
    for (int j = 0; j < N; ++j) {
      float acc = 0;
      for (int k = 0; k < K; ++k)
        acc += A[i * lda + k] * B[k + j * ldb];
      D[i * ldd + j] = alpha * acc + beta * D[i * ldd + j];
    }
}

int main() {
  int M = 1024, N = 1024, K = 512;
  int lda = K, ldb = K, ldd = N;
  float alpha = 1.0f, beta = 0.0f;

  int cA = M * K, cB = K * N, cD = M * N;
  std::vector<float> fA(cA), fB(cB), fD(cD, 0), fRef(cD, 0);
  for (int i = 0; i < cA; ++i) fA[i] = float((i * 16807 % 16) - 8) * 0.01f;
  for (int i = 0; i < cB; ++i) fB[i] = float((i * 48271 % 16) - 8) * 0.01f;

  // Convert to FP8
  std::vector<__nv_fp8_e4m3> hA(cA);
  std::vector<__nv_fp8_e5m2> hB(cB);
  std::vector<__nv_fp8_e4m3> hD(cD);
  for (int i = 0; i < cA; ++i) hA[i] = __nv_fp8_e4m3(fA[i]);
  for (int i = 0; i < cB; ++i) hB[i] = __nv_fp8_e5m2(fB[i]);

  __nv_fp8_e4m3 *dA;
  __nv_fp8_e5m2 *dB;
  __nv_fp8_e4m3 *dD;
  CUDA_CHECK(cudaMalloc(&dA, cA * sizeof(__nv_fp8_e4m3)));
  CUDA_CHECK(cudaMalloc(&dB, cB * sizeof(__nv_fp8_e5m2)));
  CUDA_CHECK(cudaMalloc(&dD, cD * sizeof(__nv_fp8_e4m3)));
  CUDA_CHECK(cudaMemcpy(dA, hA.data(), cA * sizeof(__nv_fp8_e4m3), cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(dB, hB.data(), cB * sizeof(__nv_fp8_e5m2), cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemset(dD, 0, cD * sizeof(__nv_fp8_e4m3)));

  cudaError_t r = HopperGemmPrefetch(M, N, K, alpha, dA, lda, dB, ldb, beta, dD, ldd);
  if (r != cudaSuccess) {
    std::cerr << "HopperGemmPrefetch failed: " << cudaGetErrorString(r) << std::endl;
    cudaFree(dA); cudaFree(dB); cudaFree(dD);
    return -1;
  }
  CUDA_CHECK(cudaDeviceSynchronize());

  // Benchmark (triggered by KH_BENCHMARK=N env var)
  {
    int _kh_trials = kh_benchmark_trials();
    if (_kh_trials > 0 && r == cudaSuccess) {
      kh_benchmark([&]() { HopperGemmPrefetch(M, N, K, alpha, dA, lda, dB, ldb, beta, dD, ldd); }, _kh_trials);
    }
  }


  CUDA_CHECK(cudaMemcpy(hD.data(), dD, cD * sizeof(__nv_fp8_e4m3), cudaMemcpyDeviceToHost));
  cudaFree(dA); cudaFree(dB); cudaFree(dD);

  for (int i = 0; i < cD; ++i) fD[i] = float(hD[i]);

  // Reference uses the FP8-quantized inputs (to match precision)
  std::vector<float> fA_q(cA), fB_q(cB);
  for (int i = 0; i < cA; ++i) fA_q[i] = float(hA[i]);
  for (int i = 0; i < cB; ++i) fB_q[i] = float(hB[i]);
  reference(M, N, K, alpha, fA_q, lda, fB_q, ldb, beta, fRef, ldd);

  float md = 0;
  for (int i = 0; i < cD; ++i) md = fmaxf(md, fabsf(fD[i] - fRef[i]));
  // FP8 precision: moderate tolerance
  if (md > 1.0f) {
    std::cerr << "Results incorrect. Max diff: " << md << std::endl;
    return -1;
  }
  std::cout << "Passed." << std::endl;
  return 0;
}
