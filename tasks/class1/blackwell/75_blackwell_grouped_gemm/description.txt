# Task: 75_blackwell_grouped_gemm

## Target Hardware
NVIDIA Blackwell GPU (SM100/SM120).


Blackwell SM100 Grouped GEMM test harness — FP8 precision
Based on CUTLASS example 75_blackwell_grouped_gemm
The LLM must implement:
  cudaError_t BlackwellGroupedGemm(int num_groups,
      int const *Ms, int const *Ns, int const *Ks,
      float alpha, float beta,
      __nv_fp8_e4m3 const * const *A_ptrs, int const *ldas,
      __nv_fp8_e4m3 const * const *B_ptrs, int const *ldbs,
      __half * const *D_ptrs, int const *ldds);
Each group g: D_g = alpha * A_g * B_g + beta * D_g
A_g is row-major [Ms[g], Ks[g]] (ldas[g] = Ks[g]) in FP8 E4M3,
B_g is column-major [Ks[g], Ns[g]] (ldbs[g] = Ks[g]) in FP8 E4M3,
D_g is column-major [Ms[g], Ns[g]] (ldds[g] = Ms[g]) in FP16.
FP32 accumulation.
Compile: nvcc -O3 --use_fast_math -std=c++17 --expt-relaxed-constexpr \
         -gencode arch=compute_100a,code=sm_100a harness.cu solution.cu -o test

#include <iostream>
#include <sstream>
#include <vector>
#include <cmath>
#include <cstdlib>
#include <cuda_fp8.h>
#include <cuda_fp16.h>
#include "../../include/kh_benchmark.h"

#define CUDA_CHECK(status)                                              \
  {                                                                     \
    cudaError_t error = status;                                         \
    if (error != cudaSuccess) {                                         \
      std::cerr << "Got bad cuda status: " << cudaGetErrorString(error) \
                << " at line: " << __LINE__ << std::endl;               \
      exit(EXIT_FAILURE);                                               \
    }                                                                   \
  }

extern cudaError_t BlackwellGroupedGemm(
  int num_groups,
  int const *Ms, int const *Ns, int const *Ks,
  float alpha, float beta,
  __nv_fp8_e4m3 const * const *A_ptrs, int const *ldas,
  __nv_fp8_e4m3 const * const *B_ptrs, int const *ldbs,
  __half * const *D_ptrs, int const *ldds);


// Naive reference on GPU: A row-major FP8, B col-major FP8, D col-major FP16
__global__ void ReferenceGemm_kernel(
  int M, int N, int K,
  float alpha,
  __nv_fp8_e4m3 const *A, int lda,
  __nv_fp8_e4m3 const *B, int ldb,
  float beta,
  __half *D, int ldd) {

  int i = threadIdx.x + blockIdx.x * blockDim.x;
  int j = threadIdx.y + blockIdx.y * blockDim.y;

  if (i < M && j < N) {
    float acc = 0.0f;
    for (int k = 0; k < K; ++k) {
      acc += float(A[i * lda + k]) * float(B[k + j * ldb]);
    }
    D[i + j * ldd] = __float2half(alpha * acc + beta * __half2float(D[i + j * ldd]));
  }
}

__global__ void InitFp8_kernel(__nv_fp8_e4m3 *matrix, int size, int seed) {
  int idx = threadIdx.x + blockIdx.x * blockDim.x;
  if (idx < size) {
    float value = float(((idx + seed) * 16807 % 16) - 8) * 0.125f;
    matrix[idx] = __nv_fp8_e4m3(value);
  }
}

cudaError_t TestGroupedGemm(int M, int N, int K, int num_groups, float alpha, float beta) {
  std::vector<int> Ms(num_groups, M), Ns(num_groups, N), Ks(num_groups, K);
  std::vector<int> ldas(num_groups, K), ldbs(num_groups, K), ldds(num_groups, M);

  std::vector<__nv_fp8_e4m3*> h_A_ptrs(num_groups), h_B_ptrs(num_groups);
  std::vector<__half*> h_D_test_ptrs(num_groups), h_D_ref_ptrs(num_groups);

  for (int g = 0; g < num_groups; ++g) {
    CUDA_CHECK(cudaMalloc(&h_A_ptrs[g], sizeof(__nv_fp8_e4m3) * M * K));
    CUDA_CHECK(cudaMalloc(&h_B_ptrs[g], sizeof(__nv_fp8_e4m3) * K * N));
    CUDA_CHECK(cudaMalloc(&h_D_test_ptrs[g], sizeof(__half) * M * N));
    CUDA_CHECK(cudaMalloc(&h_D_ref_ptrs[g], sizeof(__half) * M * N));

    int blocks_a = (M * K + 255) / 256;
    InitFp8_kernel<<<blocks_a, 256>>>(h_A_ptrs[g], M * K, g * 7);
    int blocks_b = (K * N + 255) / 256;
    InitFp8_kernel<<<blocks_b, 256>>>(h_B_ptrs[g], K * N, g * 13 + 17);
    CUDA_CHECK(cudaMemset(h_D_test_ptrs[g], 0, sizeof(__half) * M * N));
    CUDA_CHECK(cudaMemset(h_D_ref_ptrs[g], 0, sizeof(__half) * M * N));
  }
  CUDA_CHECK(cudaGetLastError());

  int *d_Ms, *d_Ns, *d_Ks, *d_ldas, *d_ldbs, *d_ldds;
  __nv_fp8_e4m3 **d_A_ptrs, **d_B_ptrs;
  __half **d_D_test_ptrs;
  CUDA_CHECK(cudaMalloc(&d_Ms, sizeof(int) * num_groups));
  CUDA_CHECK(cudaMalloc(&d_Ns, sizeof(int) * num_groups));
  CUDA_CHECK(cudaMalloc(&d_Ks, sizeof(int) * num_groups));
  CUDA_CHECK(cudaMalloc(&d_ldas, sizeof(int) * num_groups));
  CUDA_CHECK(cudaMalloc(&d_ldbs, sizeof(int) * num_groups));
  CUDA_CHECK(cudaMalloc(&d_ldds, sizeof(int) * num_groups));
  CUDA_CHECK(cudaMalloc(&d_A_ptrs, sizeof(__nv_fp8_e4m3*) * num_groups));
  CUDA_CHECK(cudaMalloc(&d_B_ptrs, sizeof(__nv_fp8_e4m3*) * num_groups));
  CUDA_CHECK(cudaMalloc(&d_D_test_ptrs, sizeof(__half*) * num_groups));

  CUDA_CHECK(cudaMemcpy(d_Ms, Ms.data(), sizeof(int) * num_groups, cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(d_Ns, Ns.data(), sizeof(int) * num_groups, cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(d_Ks, Ks.data(), sizeof(int) * num_groups, cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(d_ldas, ldas.data(), sizeof(int) * num_groups, cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(d_ldbs, ldbs.data(), sizeof(int) * num_groups, cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(d_ldds, ldds.data(), sizeof(int) * num_groups, cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(d_A_ptrs, h_A_ptrs.data(), sizeof(__nv_fp8_e4m3*) * num_groups, cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(d_B_ptrs, h_B_ptrs.data(), sizeof(__nv_fp8_e4m3*) * num_groups, cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(d_D_test_ptrs, h_D_test_ptrs.data(), sizeof(__half*) * num_groups, cudaMemcpyHostToDevice));

  cudaError_t result = BlackwellGroupedGemm(
    num_groups, d_Ms, d_Ns, d_Ks, alpha, beta,
    (__nv_fp8_e4m3 const * const *)d_A_ptrs, d_ldas,
    (__nv_fp8_e4m3 const * const *)d_B_ptrs, d_ldbs,
    d_D_test_ptrs, d_ldds);

  if (result != cudaSuccess) {
    std::cerr << "Grouped GEMM kernel failed: " << cudaGetErrorString(result) << std::endl;
    goto cleanup;
  }

  // Run reference per group
  for (int g = 0; g < num_groups; ++g) {
    dim3 block(16, 16);
    dim3 grid((M + 15) / 16, (N + 15) / 16);
    ReferenceGemm_kernel<<<grid, block>>>(M, N, K, alpha,
      h_A_ptrs[g], K, h_B_ptrs[g], K, beta, h_D_ref_ptrs[g], M);
  }
  CUDA_CHECK(cudaGetLastError());
  CUDA_CHECK(cudaDeviceSynchronize());

  // Benchmark (triggered by KH_BENCHMARK=N env var)
  {
    int _kh_trials = kh_benchmark_trials();
    if (_kh_trials > 0 && result == cudaSuccess) {
      kh_benchmark([&]() { BlackwellGroupedGemm(
    num_groups, d_Ms, d_Ns, d_Ks, alpha, beta,
    (__nv_fp8_e4m3 const * const *)d_A_ptrs, d_ldas,
    (__nv_fp8_e4m3 const * const *)d_B_ptrs, d_ldbs,
    d_D_test_ptrs, d_ldds); }, _kh_trials);
    }
  }
  // Benchmark reference kernel
  {
    int _kh_ref_trials = kh_benchmark_trials();
    if (_kh_ref_trials > 0) {
      kh_benchmark([&]() {
        for (int g = 0; g < num_groups; ++g) {
            dim3 block(16, 16);
            dim3 grid((M + 15) / 16, (N + 15) / 16);
            ReferenceGemm_kernel<<<grid, block>>>(M, N, K, alpha,
              h_A_ptrs[g], K, h_B_ptrs[g], K, beta, h_D_ref_ptrs[g], M);
          }
      }, _kh_ref_trials, "Ref");
    }
  }



  {
    int total_mismatches = 0;
    int total_elements = 0;
    for (int g = 0; g < num_groups; ++g) {
      std::vector<__half> h_test(M * N), h_ref(M * N);
      CUDA_CHECK(cudaMemcpy(h_test.data(), h_D_test_ptrs[g], sizeof(__half) * M * N, cudaMemcpyDeviceToHost));
      CUDA_CHECK(cudaMemcpy(h_ref.data(), h_D_ref_ptrs[g], sizeof(__half) * M * N, cudaMemcpyDeviceToHost));

      for (int idx = 0; idx < M * N; ++idx) {
        float t = __half2float(h_test[idx]);
        float r = __half2float(h_ref[idx]);
        float diff = std::fabs(t - r);
        float denom = std::fmax(std::fabs(r), 1e-6f);
        if (diff / denom > 0.3f) total_mismatches++;
      }
      total_elements += M * N;
    }

    float mismatch_rate = float(total_mismatches) / float(total_elements);
    if (mismatch_rate > 0.01f) {
      std::cerr << "Results incorrect. Mismatch rate: " << mismatch_rate << std::endl;
      result = cudaErrorUnknown;
    }
  }

cleanup:
  cudaFree(d_Ms); cudaFree(d_Ns); cudaFree(d_Ks);
  cudaFree(d_ldas); cudaFree(d_ldbs); cudaFree(d_ldds);
  cudaFree(d_A_ptrs); cudaFree(d_B_ptrs); cudaFree(d_D_test_ptrs);
  for (int g = 0; g < num_groups; ++g) {
    cudaFree(h_A_ptrs[g]); cudaFree(h_B_ptrs[g]);
    cudaFree(h_D_test_ptrs[g]); cudaFree(h_D_ref_ptrs[g]);
  }
  return result;
}

int main(int argc, const char *arg[]) {
  int M = 512, N = 512, K = 512, groups = 4;
  if (argc > 1) { std::stringstream ss(arg[1]); ss >> M; }
  if (argc > 2) { std::stringstream ss(arg[2]); ss >> N; }
  if (argc > 3) { std::stringstream ss(arg[3]); ss >> K; }
  if (argc > 4) { std::stringstream ss(arg[4]); ss >> groups; }

  cudaError_t result = TestGroupedGemm(M, N, K, groups, 1.0f, 0.0f);
  if (result == cudaSuccess) {
    std::cout << "Passed." << std::endl;
  }
  return result == cudaSuccess ? 0 : -1;
}
