# Task: 80_blackwell_geforce_sparse_gemm

## Target Hardware
NVIDIA Blackwell GPU (SM100/SM120).


Blackwell GeForce SM120 sparse GEMM test harness — 2:4 structured sparsity, MXFP8->BF16
Based on CUTLASS example 80_blackwell_geforce_sparse_gemm
The LLM must implement:
  cudaError_t BlackwellGeForceSparseMxfp8Gemm(int M, int N, int K, float alpha,
      __nv_fp8_e4m3 const *A_compressed, int lda_comp,
      float const *A_scales, int scale_lda,
      uint16_t const *A_metadata,
      __nv_fp8_e4m3 const *B, int ldb,
      float const *B_scales, int scale_ldb,
      float beta, __nv_bfloat16 *D, int ldd);
A is 2:4 sparse compressed: A_compressed is row-major M x (K/2) in FP8 E4M3.
A_metadata encodes which 2 of every 4 elements are non-zero.
A_scales: [M, ceil(K/32)] row-major, per-block scale factors.
B is column-major [K, N] in FP8 E4M3, B_scales: [N, ceil(K/32)] row-major.
D is row-major [M, N] in BF16.
Compile: nvcc -O3 --use_fast_math -std=c++17 --expt-relaxed-constexpr \
         -I<cutlass>/include \
         -gencode arch=compute_120a,code=sm_120a harness.cu solution.cu -o test

/*
 * Blackwell GeForce SM120 sparse GEMM test harness — 2:4 structured sparsity, MXFP8->BF16
 * Based on CUTLASS example 80_blackwell_geforce_sparse_gemm
 *
 * The LLM must implement:
 *   cudaError_t BlackwellGeForceSparseMxfp8Gemm(int M, int N, int K, float alpha,
 *       __nv_fp8_e4m3 const *A_compressed, int lda_comp,
 *       float const *A_scales, int scale_lda,
 *       uint16_t const *A_metadata,
 *       __nv_fp8_e4m3 const *B, int ldb,
 *       float const *B_scales, int scale_ldb,
 *       float beta, __nv_bfloat16 *D, int ldd);
 *
 * A is 2:4 sparse compressed: A_compressed is row-major M x (K/2) in FP8 E4M3.
 * A_metadata encodes which 2 of every 4 elements are non-zero.
 * A_scales: [M, ceil(K/32)] row-major, per-block scale factors.
 * B is column-major [K, N] in FP8 E4M3, B_scales: [N, ceil(K/32)] row-major.
 * D is row-major [M, N] in BF16.
 *
 * Compile: nvcc -O3 --use_fast_math -std=c++17 --expt-relaxed-constexpr \
 *          -I<cutlass>/include \
 *          -gencode arch=compute_120a,code=sm_120a harness.cu solution.cu -o test

#include <cuda_bf16.h>
#include <cuda_fp8.h>
#include <iostream>
#include <vector>
#include <cstdint>
#include <cmath>
#include "../../include/kh_benchmark.h"

#define CUDA_CHECK(status)                                              \
  {                                                                     \
    cudaError_t error = status;                                         \
    if (error != cudaSuccess) {                                         \
      std::cerr << "Got bad cuda status: " << cudaGetErrorString(error) \
                << " at line: " << __LINE__ << std::endl;               \
      exit(EXIT_FAILURE);                                               \
    }                                                                   \
  }

static constexpr int kGroupSize = 32;

extern cudaError_t BlackwellGeForceSparseMxfp8Gemm(
  int M, int N, int K, float alpha,
  __nv_fp8_e4m3 const *A_compressed, int lda_comp,
  float const *A_scales, int scale_lda,
  uint16_t const *A_metadata,
  __nv_fp8_e4m3 const *B, int ldb,
  float const *B_scales, int scale_ldb,
  float beta,
  __nv_bfloat16 *D, int ldd);

/// CPU reference: dense GEMM (with scales=1.0, comparing against sparse+dense product)
void reference_gemm(int M, int N, int K, float alpha,
  std::vector<float> const &A, int lda,
  std::vector<float> const &B, int ldb,
  float beta, std::vector<float> &D, int ldd) {
  for (int i = 0; i < M; ++i)
    for (int j = 0; j < N; ++j) {
      float acc = 0;
      for (int k = 0; k < K; ++k) acc += A[i*lda+k] * B[k+j*ldb];
      D[i*ldd+j] = alpha * acc + beta * D[i*ldd+j];
    }
}

void create_dense_2_4(std::vector<float> &dense, int M, int K) {
  for (int i = 0; i < M; ++i)
    for (int k = 0; k < K; ++k) {
      int pos = k % 4;
      if (pos == 1 || pos == 3)
        dense[i*K+k] = 0.0f;
      else
        dense[i*K+k] = float(((i*K+k) * 16807 % 16) - 8) * 0.125f;
    }
}

void create_sparse_2_4_fp8(
  std::vector<float> const &dense, int M, int K,
  std::vector<__nv_fp8_e4m3> &compressed, std::vector<uint16_t> &metadata) {
  int K_comp = K / 2;
  compressed.resize(M * K_comp);
  int meta_cols = K / 16;
  if (meta_cols == 0) meta_cols = 1;
  metadata.resize(M * meta_cols, 0);

  for (int i = 0; i < M; ++i) {
    int ci = 0;
    for (int kg = 0; kg < K; kg += 4) {
      int idx0 = 0, idx1 = 2;
      compressed[i*K_comp+ci]   = __nv_fp8_e4m3(dense[i*K+kg+idx0]);
      compressed[i*K_comp+ci+1] = __nv_fp8_e4m3(dense[i*K+kg+idx1]);
      ci += 2;
      int group = kg / 4;
      int meta_word = group / 4;
      int meta_shift = (group % 4) * 4;
      uint16_t bits = (uint16_t)((idx0 & 0x3) | ((idx1 & 0x3) << 2));
      metadata[i*meta_cols+meta_word] |= (bits << meta_shift);
    }
  }
}

int main() {
  int M = 128, N = 128, K = 256;
  int ldb = K, ldd = N, lda_comp = K / 2;
  float alpha = 1.0f, beta = 0.0f;
  int k_groups = (K + kGroupSize - 1) / kGroupSize;

  std::vector<float> fA_dense(M * K), fB_raw(K * N);
  create_dense_2_4(fA_dense, M, K);
  for (int i = 0; i < K * N; ++i) fB_raw[i] = float((i * 48271 % 16) - 8) * 0.125f;

  // Create FP8 compressed A and FP8 B
  std::vector<__nv_fp8_e4m3> hA_comp;
  std::vector<uint16_t> h_meta;
  create_sparse_2_4_fp8(fA_dense, M, K, hA_comp, h_meta);

  std::vector<__nv_fp8_e4m3> hB(K * N);
  std::vector<float> fB(K * N);
  for (int i = 0; i < K * N; ++i) { hB[i] = __nv_fp8_e4m3(fB_raw[i]); fB[i] = float(hB[i]); }

  // Reconstruct dense A from compressed FP8 for reference
  std::vector<float> fA_ref(M * K, 0);
  int K_comp = K / 2;
  for (int i = 0; i < M; ++i) {
    int ci = 0;
    for (int kg = 0; kg < K; kg += 4) {
      fA_ref[i*K+kg+0] = float(hA_comp[i*K_comp+ci]);
      fA_ref[i*K+kg+2] = float(hA_comp[i*K_comp+ci+1]);
      ci += 2;
    }
  }

  std::vector<float> sA(M * k_groups, 1.0f), sB(N * k_groups, 1.0f);
  int meta_cols = K / 16; if (meta_cols == 0) meta_cols = 1;

  __nv_fp8_e4m3 *dA_comp, *dB; uint16_t *dA_meta;
  float *dsA, *dsB; __nv_bfloat16 *dD;
  CUDA_CHECK(cudaMalloc(&dA_comp, M*K_comp*sizeof(__nv_fp8_e4m3)));
  CUDA_CHECK(cudaMalloc(&dA_meta, M*meta_cols*sizeof(uint16_t)));
  CUDA_CHECK(cudaMalloc(&dsA, M*k_groups*sizeof(float)));
  CUDA_CHECK(cudaMalloc(&dB, K*N*sizeof(__nv_fp8_e4m3)));
  CUDA_CHECK(cudaMalloc(&dsB, N*k_groups*sizeof(float)));
  CUDA_CHECK(cudaMalloc(&dD, M*N*sizeof(__nv_bfloat16)));

  CUDA_CHECK(cudaMemcpy(dA_comp, hA_comp.data(), M*K_comp*sizeof(__nv_fp8_e4m3), cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(dA_meta, h_meta.data(), M*meta_cols*sizeof(uint16_t), cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(dsA, sA.data(), M*k_groups*sizeof(float), cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(dB, hB.data(), K*N*sizeof(__nv_fp8_e4m3), cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(dsB, sB.data(), N*k_groups*sizeof(float), cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemset(dD, 0, M*N*sizeof(__nv_bfloat16)));

  cudaError_t r = BlackwellGeForceSparseMxfp8Gemm(
    M, N, K, alpha, dA_comp, lda_comp, dsA, k_groups, dA_meta, dB, ldb, dsB, k_groups, beta, dD, ldd);
  if (r != cudaSuccess) {
    std::cerr << "Sparse GEMM failed: " << cudaGetErrorString(r) << std::endl;
    cudaFree(dA_comp); cudaFree(dA_meta); cudaFree(dsA); cudaFree(dB); cudaFree(dsB); cudaFree(dD);
    return -1;
  }
  CUDA_CHECK(cudaDeviceSynchronize());

  // Benchmark (triggered by KH_BENCHMARK=N env var)
  {
    int _kh_trials = kh_benchmark_trials();
    if (_kh_trials > 0 && r == cudaSuccess) {
      kh_benchmark([&]() { BlackwellGeForceSparseMxfp8Gemm(
    M, N, K, alpha, dA_comp, lda_comp, dsA, k_groups, dA_meta, dB, ldb, dsB, k_groups, beta, dD, ldd); }, _kh_trials);
    }
  }


  std::vector<__nv_bfloat16> hD(M*N);
  CUDA_CHECK(cudaMemcpy(hD.data(), dD, M*N*sizeof(__nv_bfloat16), cudaMemcpyDeviceToHost));
  cudaFree(dA_comp); cudaFree(dA_meta); cudaFree(dsA); cudaFree(dB); cudaFree(dsB); cudaFree(dD);

  std::vector<float> fD(M*N), fRef(M*N, 0);
  for (int i = 0; i < M*N; ++i) fD[i] = __bfloat162float(hD[i]);
  reference_gemm(M, N, K, alpha, fA_ref, K, fB, ldb, beta, fRef, ldd);

  float max_rel = 0; int mm = 0;
  for (int i = 0; i < M*N; ++i) {
    float d = std::fabs(fD[i]-fRef[i]), den = std::fmax(std::fabs(fRef[i]),1e-6f);
    float rel = d/den; max_rel = std::fmax(max_rel, rel);
    if (rel > 0.3f) mm++;
  }
  if (float(mm)/float(M*N) > 0.01f) {
    std::cerr << "Results incorrect. Max relative diff: " << max_rel
              << ", mismatch rate: " << float(mm)/float(M*N) << std::endl;
    return -1;
  }
  std::cout << "Passed." << std::endl;
  return 0;
}
