# Task: 79d_blackwell_geforce_nvfp4_grouped_gemm

## Target Hardware
NVIDIA Blackwell GPU (SM100/SM120).


Blackwell GeForce SM120 NVFP4 Grouped GEMM test harness
Based on CUTLASS example 79d_blackwell_geforce_nvfp4_grouped_gemm
The LLM must implement:
  cudaError_t Nvfp4GroupedGemm(int num_groups,
      int const *Ms, int const *Ns, int const *Ks,
      float alpha, float beta,
      cutlass::float_e2m1_t const * const *A_ptrs, int const *ldas,
      float const * const *A_scale_ptrs, int const *scale_ldas,
      cutlass::float_e2m1_t const * const *B_ptrs, int const *ldbs,
      float const * const *B_scale_ptrs, int const *scale_ldbs,
      cutlass::float_e2m1_t * const *D_ptrs, int const *ldds,
      float * const *D_scale_ptrs, int const *scale_ldds);
Each group g: D_g = alpha * A_g * B_g + beta * D_g
A_g row-major [Ms[g], Ks[g]] FP4, B_g col-major [Ks[g], Ns[g]] FP4,
D_g row-major [Ms[g], Ns[g]] FP4 with output scale factors.
One element per byte (unpacked). Scale group size = 32, output SF vec = 16.
Compile: nvcc -O3 --use_fast_math -std=c++17 --expt-relaxed-constexpr \
         -I<cutlass>/include \
         -gencode arch=compute_120a,code=sm_120a harness.cu solution.cu -o test

/*
 * Blackwell GeForce SM120 NVFP4 Grouped GEMM test harness
 * Based on CUTLASS example 79d_blackwell_geforce_nvfp4_grouped_gemm
 *
 * The LLM must implement:
 *   cudaError_t Nvfp4GroupedGemm(int num_groups,
 *       int const *Ms, int const *Ns, int const *Ks,
 *       float alpha, float beta,
 *       cutlass::float_e2m1_t const * const *A_ptrs, int const *ldas,
 *       float const * const *A_scale_ptrs, int const *scale_ldas,
 *       cutlass::float_e2m1_t const * const *B_ptrs, int const *ldbs,
 *       float const * const *B_scale_ptrs, int const *scale_ldbs,
 *       cutlass::float_e2m1_t * const *D_ptrs, int const *ldds,
 *       float * const *D_scale_ptrs, int const *scale_ldds);
 *
 * Each group g: D_g = alpha * A_g * B_g + beta * D_g
 * A_g row-major [Ms[g], Ks[g]] FP4, B_g col-major [Ks[g], Ns[g]] FP4,
 * D_g row-major [Ms[g], Ns[g]] FP4 with output scale factors.
 * One element per byte (unpacked). Scale group size = 32, output SF vec = 16.
 *
 * Compile: nvcc -O3 --use_fast_math -std=c++17 --expt-relaxed-constexpr \
 *          -I<cutlass>/include \
 *          -gencode arch=compute_120a,code=sm_120a harness.cu solution.cu -o test

#include <iostream>
#include <sstream>
#include <vector>
#include <cmath>
#include <cstdlib>
#include "cutlass/numeric_types.h"
#include "../../include/kh_benchmark.h"

#define CUDA_CHECK(status)                                              \
  {                                                                     \
    cudaError_t error = status;                                         \
    if (error != cudaSuccess) {                                         \
      std::cerr << "Got bad cuda status: " << cudaGetErrorString(error) \
                << " at line: " << __LINE__ << std::endl;               \
      exit(EXIT_FAILURE);                                               \
    }                                                                   \
  }

static constexpr int kGroupSize = 32;
static constexpr int kOutputSFVec = 16;

extern cudaError_t Nvfp4GroupedGemm(
  int num_groups,
  int const *Ms, int const *Ns, int const *Ks,
  float alpha, float beta,
  cutlass::float_e2m1_t const * const *A_ptrs, int const *ldas,
  float const * const *A_scale_ptrs, int const *scale_ldas,
  cutlass::float_e2m1_t const * const *B_ptrs, int const *ldbs,
  float const * const *B_scale_ptrs, int const *scale_ldbs,
  cutlass::float_e2m1_t * const *D_ptrs, int const *ldds,
  float * const *D_scale_ptrs, int const *scale_ldds);

void reference_group(int M, int N, int K, float alpha,
  std::vector<float> const &fA, int lda,
  std::vector<float> const &sA, int scale_lda,
  std::vector<float> const &fB, int ldb,
  std::vector<float> const &sB, int scale_ldb,
  float beta, std::vector<float> &D, int ldd) {
  for (int i = 0; i < M; ++i)
    for (int j = 0; j < N; ++j) {
      float acc = 0;
      for (int k = 0; k < K; ++k) {
        int kg = k / kGroupSize;
        acc += fA[i*lda+k] * sA[i*scale_lda+kg] * fB[k+j*ldb] * sB[j*scale_ldb+kg];
      }
      D[i*ldd+j] = alpha * acc + beta * D[i*ldd+j];
    }
}

int main(int argc, const char *arg[]) {
  int M = 256, N = 256, K = 256, num_groups = 4;
  if (argc > 1) { std::stringstream ss(arg[1]); ss >> M; }
  if (argc > 2) { std::stringstream ss(arg[2]); ss >> N; }
  if (argc > 3) { std::stringstream ss(arg[3]); ss >> K; }
  if (argc > 4) { std::stringstream ss(arg[4]); ss >> num_groups; }

  float alpha = 1.0f, beta = 0.0f;
  int k_groups = (K+kGroupSize-1)/kGroupSize;
  int n_sf = (N+kOutputSFVec-1)/kOutputSFVec;
  std::vector<int> Ms(num_groups, M), Ns(num_groups, N), Ks(num_groups, K);
  std::vector<int> ldas(num_groups, K), ldbs(num_groups, K), ldds(num_groups, N);
  std::vector<int> sldas(num_groups, k_groups), sldbs(num_groups, k_groups), sldds(num_groups, n_sf);

  std::vector<std::vector<float>> fAs(num_groups), fBs(num_groups), sAs(num_groups), sBs(num_groups);
  std::vector<cutlass::float_e2m1_t*> dA(num_groups), dB(num_groups), dD(num_groups);
  std::vector<float*> dsA(num_groups), dsB(num_groups), dsD(num_groups);

  for (int g = 0; g < num_groups; ++g) {
    std::vector<cutlass::float_e2m1_t> hA(M*K), hB(K*N);
    fAs[g].resize(M*K); fBs[g].resize(K*N);
    sAs[g].assign(M*k_groups, 1.0f); sBs[g].assign(N*k_groups, 1.0f);
    for (int i = 0; i < M*K; ++i) {
      hA[i] = cutlass::float_e2m1_t(float(((i+g*7)*16807%6)-3)*0.5f);
      fAs[g][i] = float(hA[i]);
    }
    for (int i = 0; i < K*N; ++i) {
      hB[i] = cutlass::float_e2m1_t(float(((i+g*13)*48271%6)-3)*0.5f);
      fBs[g][i] = float(hB[i]);
    }
    CUDA_CHECK(cudaMalloc(&dA[g], M*K*sizeof(cutlass::float_e2m1_t)));
    CUDA_CHECK(cudaMalloc(&dB[g], K*N*sizeof(cutlass::float_e2m1_t)));
    CUDA_CHECK(cudaMalloc(&dsA[g], M*k_groups*sizeof(float)));
    CUDA_CHECK(cudaMalloc(&dsB[g], N*k_groups*sizeof(float)));
    CUDA_CHECK(cudaMalloc(&dD[g], M*N*sizeof(cutlass::float_e2m1_t)));
    CUDA_CHECK(cudaMalloc(&dsD[g], M*n_sf*sizeof(float)));
    CUDA_CHECK(cudaMemcpy(dA[g], hA.data(), M*K*sizeof(cutlass::float_e2m1_t), cudaMemcpyHostToDevice));
    CUDA_CHECK(cudaMemcpy(dB[g], hB.data(), K*N*sizeof(cutlass::float_e2m1_t), cudaMemcpyHostToDevice));
    CUDA_CHECK(cudaMemcpy(dsA[g], sAs[g].data(), M*k_groups*sizeof(float), cudaMemcpyHostToDevice));
    CUDA_CHECK(cudaMemcpy(dsB[g], sBs[g].data(), N*k_groups*sizeof(float), cudaMemcpyHostToDevice));
    CUDA_CHECK(cudaMemset(dD[g], 0, M*N*sizeof(cutlass::float_e2m1_t)));
    CUDA_CHECK(cudaMemset(dsD[g], 0, M*n_sf*sizeof(float)));
  }

  // Device arrays of pointers/dims
  int *d_Ms, *d_Ns, *d_Ks, *d_ldas, *d_ldbs, *d_ldds, *d_sldas, *d_sldbs, *d_sldds;
  cutlass::float_e2m1_t **d_Ap, **d_Bp, **d_Dp; float **d_sAp, **d_sBp, **d_sDp;
  CUDA_CHECK(cudaMalloc(&d_Ms, sizeof(int)*num_groups)); CUDA_CHECK(cudaMalloc(&d_Ns, sizeof(int)*num_groups));
  CUDA_CHECK(cudaMalloc(&d_Ks, sizeof(int)*num_groups)); CUDA_CHECK(cudaMalloc(&d_ldas, sizeof(int)*num_groups));
  CUDA_CHECK(cudaMalloc(&d_ldbs, sizeof(int)*num_groups)); CUDA_CHECK(cudaMalloc(&d_ldds, sizeof(int)*num_groups));
  CUDA_CHECK(cudaMalloc(&d_sldas, sizeof(int)*num_groups)); CUDA_CHECK(cudaMalloc(&d_sldbs, sizeof(int)*num_groups));
  CUDA_CHECK(cudaMalloc(&d_sldds, sizeof(int)*num_groups));
  CUDA_CHECK(cudaMalloc(&d_Ap, sizeof(void*)*num_groups)); CUDA_CHECK(cudaMalloc(&d_Bp, sizeof(void*)*num_groups));
  CUDA_CHECK(cudaMalloc(&d_Dp, sizeof(void*)*num_groups)); CUDA_CHECK(cudaMalloc(&d_sAp, sizeof(void*)*num_groups));
  CUDA_CHECK(cudaMalloc(&d_sBp, sizeof(void*)*num_groups)); CUDA_CHECK(cudaMalloc(&d_sDp, sizeof(void*)*num_groups));
  CUDA_CHECK(cudaMemcpy(d_Ms, Ms.data(), sizeof(int)*num_groups, cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(d_Ns, Ns.data(), sizeof(int)*num_groups, cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(d_Ks, Ks.data(), sizeof(int)*num_groups, cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(d_ldas, ldas.data(), sizeof(int)*num_groups, cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(d_ldbs, ldbs.data(), sizeof(int)*num_groups, cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(d_ldds, ldds.data(), sizeof(int)*num_groups, cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(d_sldas, sldas.data(), sizeof(int)*num_groups, cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(d_sldbs, sldbs.data(), sizeof(int)*num_groups, cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(d_sldds, sldds.data(), sizeof(int)*num_groups, cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(d_Ap, dA.data(), sizeof(void*)*num_groups, cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(d_Bp, dB.data(), sizeof(void*)*num_groups, cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(d_Dp, dD.data(), sizeof(void*)*num_groups, cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(d_sAp, dsA.data(), sizeof(void*)*num_groups, cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(d_sBp, dsB.data(), sizeof(void*)*num_groups, cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(d_sDp, dsD.data(), sizeof(void*)*num_groups, cudaMemcpyHostToDevice));

  cudaError_t result = Nvfp4GroupedGemm(num_groups, d_Ms, d_Ns, d_Ks, alpha, beta,
    (cutlass::float_e2m1_t const*const*)d_Ap, d_ldas, (float const*const*)d_sAp, d_sldas,
    (cutlass::float_e2m1_t const*const*)d_Bp, d_ldbs, (float const*const*)d_sBp, d_sldbs,
    d_Dp, d_ldds, d_sDp, d_sldds);

  if (result != cudaSuccess) {
    std::cerr << "Grouped GEMM failed: " << cudaGetErrorString(result) << std::endl;
  } else {
    CUDA_CHECK(cudaDeviceSynchronize());

  // Benchmark (triggered by KH_BENCHMARK=N env var)
  {
    int _kh_trials = kh_benchmark_trials();
    if (_kh_trials > 0 && result == cudaSuccess) {
      kh_benchmark([&]() { Nvfp4GroupedGemm(num_groups, d_Ms, d_Ns, d_Ks, alpha, beta,
    (cutlass::float_e2m1_t const*const*)d_Ap, d_ldas, (float const*const*)d_sAp, d_sldas,
    (cutlass::float_e2m1_t const*const*)d_Bp, d_ldbs, (float const*const*)d_sBp, d_sldbs,
    d_Dp, d_ldds, d_sDp, d_sldds); }, _kh_trials);
    }
  }

    int total_mm = 0, total_el = 0;
    for (int g = 0; g < num_groups; ++g) {
      std::vector<cutlass::float_e2m1_t> hD(M*N);
      std::vector<float> hDs(M*n_sf);
      CUDA_CHECK(cudaMemcpy(hD.data(), dD[g], M*N*sizeof(cutlass::float_e2m1_t), cudaMemcpyDeviceToHost));
      CUDA_CHECK(cudaMemcpy(hDs.data(), dsD[g], M*n_sf*sizeof(float), cudaMemcpyDeviceToHost));
      std::vector<float> fD(M*N), fRef(M*N, 0);
      for (int i = 0; i < M; ++i)
        for (int j = 0; j < N; ++j)
          fD[i*N+j] = float(hD[i*N+j]) * hDs[i*n_sf + j/kOutputSFVec];
      reference_group(M, N, K, alpha, fAs[g], K, sAs[g], k_groups, fBs[g], K, sBs[g], k_groups, beta, fRef, N);
      for (int i = 0; i < M*N; ++i) {
        float d = std::fabs(fD[i]-fRef[i]), den = std::fmax(std::fabs(fRef[i]),1e-6f);
        if (d/den > 0.5f) total_mm++;
      }
      total_el += M*N;
    }
    if (float(total_mm)/float(total_el) > 0.05f) {
      std::cerr << "Results incorrect. Mismatch rate: " << float(total_mm)/float(total_el) << std::endl;
      result = cudaErrorUnknown;
    }
  }

  cudaFree(d_Ms); cudaFree(d_Ns); cudaFree(d_Ks); cudaFree(d_ldas); cudaFree(d_ldbs); cudaFree(d_ldds);
  cudaFree(d_sldas); cudaFree(d_sldbs); cudaFree(d_sldds);
  cudaFree(d_Ap); cudaFree(d_Bp); cudaFree(d_Dp); cudaFree(d_sAp); cudaFree(d_sBp); cudaFree(d_sDp);
  for (int g = 0; g < num_groups; ++g) {
    cudaFree(dA[g]); cudaFree(dB[g]); cudaFree(dD[g]);
    cudaFree(dsA[g]); cudaFree(dsB[g]); cudaFree(dsD[g]);
  }
  if (result == cudaSuccess) std::cout << "Passed." << std::endl;
  return result == cudaSuccess ? 0 : -1;
}
