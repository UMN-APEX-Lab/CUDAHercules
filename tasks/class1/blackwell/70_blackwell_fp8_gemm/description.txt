/*
 * Blackwell SM120 FP8 GEMM test harness
 * SM120 tensor cores support F8/F6/F4 MMA (not FP16 via CUTLASS)
 *
 * The LLM must implement:
 *   cudaError_t BlackwellGeforceFp8Gemm(int M, int N, int K, float alpha,
 *       __nv_fp8_e4m3 const *A, int lda, __nv_fp8_e4m3 const *B, int ldb,
 *       float beta, __nv_bfloat16 *C, int ldc);
 *
 * A row-major [M,K], B col-major [K,N], C row-major [M,N]
 * FP8 E4M3 inputs, BF16 output, FP32 accumulation.
 *
 * Compile: nvcc -O3 harness.cu solution.cu -o test && ./test [M N K]
 */

#include <iostream>
#include <sstream>
#include <vector>
#include <cmath>
#include <cuda_fp8.h>
#include <cuda_bf16.h>
#include "../../include/kh_benchmark.h"

#define CUDA_CHECK(status)                                              \
  {                                                                     \
    cudaError_t error = status;                                         \
    if (error != cudaSuccess) {                                         \
      std::cerr << "Got bad cuda status: " << cudaGetErrorString(error) \
                << " at line: " << __LINE__ << std::endl;               \
      exit(EXIT_FAILURE);                                               \
    }                                                                   \
  }

extern cudaError_t BlackwellGeforceFp8Gemm(
  int M, int N, int K,
  float alpha,
  __nv_fp8_e4m3 const *A, int lda,
  __nv_fp8_e4m3 const *B, int ldb,
  float beta,
  __nv_bfloat16 *C, int ldc);


__global__ void InitFp8Matrix_kernel(
  __nv_fp8_e4m3 *matrix, int size, int seed) {
  int idx = threadIdx.x + blockIdx.x * blockDim.x;
  if (idx < size) {
    int const k = 16807;
    int const m = 4;  // small range for fp8
    float value = float(((idx + seed) * k % m) - m / 2);
    matrix[idx] = __nv_fp8_e4m3(value);
  }
}

// Naive reference: A row-major, B col-major, C row-major
__global__ void ReferenceFp8Gemm_kernel(
  int M, int N, int K,
  float alpha,
  __nv_fp8_e4m3 const *A, int lda,
  __nv_fp8_e4m3 const *B, int ldb,
  float beta,
  __nv_bfloat16 *C, int ldc) {

  int i = threadIdx.x + blockIdx.x * blockDim.x;
  int j = threadIdx.y + blockIdx.y * blockDim.y;

  if (i < M && j < N) {
    float acc = 0.0f;
    for (int k = 0; k < K; ++k) {
      float a_val = float(A[i * lda + k]);
      float b_val = float(B[k + j * ldb]);
      acc += a_val * b_val;
    }
    float c_val = __bfloat162float(C[i * ldc + j]);
    C[i * ldc + j] = __float2bfloat16(alpha * acc + beta * c_val);
  }
}

cudaError_t TestBlackwellGeforceFp8Gemm(int M, int N, int K, float alpha, float beta) {
  int lda = K, ldb = K, ldc = N;

  __nv_fp8_e4m3 *A, *B;
  __nv_bfloat16 *C_test, *C_ref;
  size_t sizeof_A = sizeof(__nv_fp8_e4m3) * M * K;
  size_t sizeof_B = sizeof(__nv_fp8_e4m3) * K * N;
  size_t sizeof_C = sizeof(__nv_bfloat16) * M * N;

  CUDA_CHECK(cudaMalloc(&A, sizeof_A));
  CUDA_CHECK(cudaMalloc(&B, sizeof_B));
  CUDA_CHECK(cudaMalloc(&C_test, sizeof_C));
  CUDA_CHECK(cudaMalloc(&C_ref, sizeof_C));

  {
    int blocks = (M * K + 255) / 256;
    InitFp8Matrix_kernel<<<blocks, 256>>>(A, M * K, 0);
    CUDA_CHECK(cudaGetLastError());
  }
  {
    int blocks = (K * N + 255) / 256;
    InitFp8Matrix_kernel<<<blocks, 256>>>(B, K * N, 17);
    CUDA_CHECK(cudaGetLastError());
  }
  CUDA_CHECK(cudaMemset(C_test, 0, sizeof_C));
  CUDA_CHECK(cudaMemset(C_ref, 0, sizeof_C));

  cudaError_t result = BlackwellGeforceFp8Gemm(M, N, K, alpha, A, lda, B, ldb, beta, C_test, ldc);
  if (result != cudaSuccess) {
    std::cerr << "Blackwell GeForce FP8 GEMM kernel failed" << std::endl;
    cudaFree(A); cudaFree(B); cudaFree(C_test); cudaFree(C_ref);
    return result;
  }

  {
    dim3 block(16, 16);
    dim3 grid((M + 15) / 16, (N + 15) / 16);
    ReferenceFp8Gemm_kernel<<<grid, block>>>(M, N, K, alpha, A, lda, B, ldb, beta, C_ref, ldc);
    CUDA_CHECK(cudaGetLastError());
  }
  CUDA_CHECK(cudaDeviceSynchronize());

  // Benchmark (triggered by KH_BENCHMARK=N env var)
  {
    int _kh_trials = kh_benchmark_trials();
    if (_kh_trials > 0 && result == cudaSuccess) {
      kh_benchmark([&]() { BlackwellGeforceFp8Gemm(M, N, K, alpha, A, lda, B, ldb, beta, C_test, ldc); }, _kh_trials);
    }
  }
  // Benchmark reference kernel
  {
    int _kh_ref_trials = kh_benchmark_trials();
    if (_kh_ref_trials > 0) {
      kh_benchmark([&]() {
        dim3 block(16, 16);
        dim3 grid((M + 15) / 16, (N + 15) / 16);
        ReferenceFp8Gemm_kernel<<<grid, block>>>(M, N, K, alpha, A, lda, B, ldb, beta, C_ref, ldc);
      }, _kh_ref_trials, "Ref");
    }
  }



  std::vector<__nv_bfloat16> h_test(M * N), h_ref(M * N);
  CUDA_CHECK(cudaMemcpy(h_test.data(), C_test, sizeof_C, cudaMemcpyDeviceToHost));
  CUDA_CHECK(cudaMemcpy(h_ref.data(), C_ref, sizeof_C, cudaMemcpyDeviceToHost));

  cudaFree(A); cudaFree(B); cudaFree(C_test); cudaFree(C_ref);

  float max_diff = 0.0f;
  for (int idx = 0; idx < M * N; ++idx) {
    float diff = std::fabs(__bfloat162float(h_test[idx]) - __bfloat162float(h_ref[idx]));
    max_diff = std::fmax(max_diff, diff);
  }

  // FP8 E4M3 has 3 mantissa bits — generous tolerance
  float tolerance = 2.0f;
  if (max_diff > tolerance) {
    std::cerr << "Results incorrect. Max diff: " << max_diff << std::endl;
    return cudaErrorUnknown;
  }
  return cudaSuccess;
}

int main(int argc, const char *arg[]) {
  int problem[3] = { 1024, 1024, 1024 };
  for (int i = 1; i < argc && i < 4; ++i) {
    std::stringstream ss(arg[i]);
    ss >> problem[i - 1];
  }

  cudaError_t result = TestBlackwellGeforceFp8Gemm(problem[0], problem[1], problem[2], 1.0f, 0.0f);
  if (result == cudaSuccess) {
    std::cout << "Passed." << std::endl;
  }
  return result == cudaSuccess ? 0 : -1;
}
