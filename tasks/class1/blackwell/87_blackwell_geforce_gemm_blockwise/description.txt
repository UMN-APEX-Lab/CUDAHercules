 *       __nv_fp8_e4m3 const *B, int ldb, float const *scale_B,

#include <sstream>
#include <vector>
#include <cmath>
#include <cuda_bf16.h>

#define CUDA_CHECK(status)                                              \
  {                                                                     \
    cudaError_t error = status;                                         \
    if (error != cudaSuccess) {                                         \
      std::cerr << "Got bad cuda status: " << cudaGetErrorString(error) \

extern cudaError_t BlackwellGeForceBlockwiseGemm(
  int M, int N, int K,
  float alpha,
  __nv_fp8_e4m3 const *A, int lda,
  float const *scale_A,
  __nv_fp8_e4m3 const *B, int ldb,
  float const *scale_B,
  float beta,
  __nv_bfloat16 *D, int ldd,
  int block_size);

/// CPU reference: dequantize FP8 with block scales, standard GEMM
/// A row-major, B col-major, D row-major
void reference_gemm(int M, int N, int K, float alpha,
  std::vector<float> const &fA, int lda,
  std::vector<float> const &sA,
  std::vector<float> const &fB, int ldb,
  std::vector<float> const &sB,
  float beta,
  std::vector<float> &D, int ldd,
  int block_size) {
  int k_blocks = (K + block_size - 1) / block_size;
  for (int i = 0; i < M; ++i)
    for (int j = 0; j < N; ++j) {
      float acc = 0;
      for (int k = 0; k < K; ++k) {
        int kb = k / block_size;
        float a_val = fA[i * lda + k] * sA[i * k_blocks + kb];
        float b_val = fB[k + j * ldb] * sB[j * k_blocks + kb];
        acc += a_val * b_val;
      }
      D[i * ldd + j] = alpha * acc + beta * D[i * ldd + j];
    }
}

cudaError_t TestGemm(int M, int N, int K, float alpha, float beta, int block_size) {
  int lda = K, ldb = K, ldd = N;

  int blocks_per_row = (K + block_size - 1) / block_size;
  int sA_count = M * blocks_per_row;
  int sB_count = N * blocks_per_row;

  // Host data
  std::vector<float> fA_raw(M * K), fB_raw(K * N);
  for (int i = 0; i < M * K; ++i) fA_raw[i] = float((i * 16807 % 16) - 8) * 0.125f;
  for (int i = 0; i < K * N; ++i) fB_raw[i] = float((i * 48271 % 16) - 8) * 0.125f;

  std::vector<__nv_fp8_e4m3> hA(M * K), hB(K * N);
  std::vector<float> fA(M * K), fB(K * N);
  for (int i = 0; i < M * K; ++i) { hA[i] = __nv_fp8_e4m3(fA_raw[i]); fA[i] = float(hA[i]); }
  for (int i = 0; i < K * N; ++i) { hB[i] = __nv_fp8_e4m3(fB_raw[i]); fB[i] = float(hB[i]); }

  std::vector<float> sA(sA_count, 1.0f), sB(sB_count, 1.0f);

  // Device allocations
  __nv_fp8_e4m3 *dA, *dB;
  float *dsA, *dsB;
  __nv_bfloat16 *dD;
  CUDA_CHECK(cudaMalloc(&dA, M * K * sizeof(__nv_fp8_e4m3)));
  CUDA_CHECK(cudaMalloc(&dB, K * N * sizeof(__nv_fp8_e4m3)));
  CUDA_CHECK(cudaMalloc(&dsA, sA_count * sizeof(float)));
  CUDA_CHECK(cudaMalloc(&dsB, sB_count * sizeof(float)));
  CUDA_CHECK(cudaMalloc(&dD, M * N * sizeof(__nv_bfloat16)));

  CUDA_CHECK(cudaMemcpy(dA, hA.data(), M * K * sizeof(__nv_fp8_e4m3), cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(dB, hB.data(), K * N * sizeof(__nv_fp8_e4m3), cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(dsA, sA.data(), sA_count * sizeof(float), cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemcpy(dsB, sB.data(), sB_count * sizeof(float), cudaMemcpyHostToDevice));
  CUDA_CHECK(cudaMemset(dD, 0, M * N * sizeof(__nv_bfloat16)));

  cudaError_t r = BlackwellGeForceBlockwiseGemm(
    M, N, K, alpha, dA, lda, dsA, dB, ldb, dsB, beta, dD, ldd, block_size);
  if (r != cudaSuccess) {
    std::cerr << "BlackwellGeForceBlockwiseGemm failed: " << cudaGetErrorString(r) << std::endl;
    cudaFree(dA); cudaFree(dB); cudaFree(dsA); cudaFree(dsB); cudaFree(dD);
    return r;
  }
  CUDA_CHECK(cudaDeviceSynchronize());

  // Benchmark (triggered by KH_BENCHMARK=N env var)
  {
    int _kh_trials = kh_benchmark_trials();
    if (_kh_trials > 0 && r == cudaSuccess) {
      kh_benchmark([&]() { BlackwellGeForceBlockwiseGemm(
    M, N, K, alpha, dA, lda, dsA, dB, ldb, dsB, beta, dD, ldd, block_size); }, _kh_trials);
    }
  }


  std::vector<__nv_bfloat16> hD(M * N);
  CUDA_CHECK(cudaMemcpy(hD.data(), dD, M * N * sizeof(__nv_bfloat16), cudaMemcpyDeviceToHost));
  cudaFree(dA); cudaFree(dB); cudaFree(dsA); cudaFree(dsB); cudaFree(dD);

  std::vector<float> fD(M * N), fRef(M * N, 0);
  for (int i = 0; i < M * N; ++i) fD[i] = __bfloat162float(hD[i]);
  reference_gemm(M, N, K, alpha, fA, lda, sA, fB, ldb, sB, beta, fRef, ldd, block_size);

  float max_rel_diff = 0.0f;
  int mismatches = 0;
  for (int i = 0; i < M * N; ++i) {
    float diff = std::fabs(fD[i] - fRef[i]);
    float denom = std::fmax(std::fabs(fRef[i]), 1e-6f);
    float rel = diff / denom;
    max_rel_diff = std::fmax(max_rel_diff, rel);
    if (rel > 0.3f) mismatches++;
  }

  float mismatch_rate = float(mismatches) / float(M * N);
  if (mismatch_rate > 0.01f) {
    std::cerr << "Results incorrect. Max relative diff: " << max_rel_diff
              << ", mismatch rate: " << mismatch_rate << std::endl;
    return cudaErrorUnknown;
  }
  return cudaSuccess;
}

int main(int argc, const char *arg[]) {
  int M = 256, N = 256, K = 256;
  int block_size = 128;

  for (int i = 1; i < argc && i < 4; ++i) {
    std::stringstream ss(arg[i]);
    if (i == 1) ss >> M;
    else if (i == 2) ss >> N;
    else if (i == 3) ss >> K;
  }

  cudaError_t result = TestGemm(M, N, K, 1.0f, 0.0f, block_size);
  if (result == cudaSuccess) {
    std::cout << "Passed." << std::endl;
  }
  return result == cudaSuccess ? 0 : -1;
}
