/*
 * Blackwell SM100 GEMM with Preferred Cluster Shape test harness

 *       __half const *A, int lda, __half const *B, int ldb,
 *       float beta, float *C, int ldc);
 *
 * A is row-major [M, K] (lda = K), B is column-major [K, N] (ldb = K),
 * C is column-major [M, N] (ldc = M). FP16 input, FP32 output, FP32 accumulation.
 *
 * Compile: nvcc -O3 --use_fast_math -std=c++17 --expt-relaxed-constexpr \

#include <iostream>
#include <sstream>
#include <vector>
#include <cmath>
#include <cuda_fp16.h>
#include "../../include/kh_benchmark.h"

#define CUDA_CHECK(status)                                              \
  {                                                                     \
    cudaError_t error = status;                                         \
    if (error != cudaSuccess) {                                         \
      std::cerr << "Got bad cuda status: " << cudaGetErrorString(error) \
                << " at line: " << __LINE__ << std::endl;               \
      exit(EXIT_FAILURE);                                               \
    }                                                                   \
  }

extern cudaError_t BlackwellClusterGemm(
  int M, int N, int K,
  float alpha,
  __half const *A, int lda,
  __half const *B, int ldb,
  float beta,
  float *C, int ldc);


__global__ void InitHalfMatrix_kernel(__half *matrix, int size, int seed) {
  int idx = threadIdx.x + blockIdx.x * blockDim.x;
  if (idx < size) {
    int const k = 16807;
    int const m = 8;
    float value = float(((idx + seed) * k % m) - m / 2);
    matrix[idx] = __float2half(value);
  }
}

// Naive reference: A row-major, B col-major, C col-major, fp32 output
__global__ void ReferenceGemm_kernel(
  int M, int N, int K,
  float alpha,
  __half const *A, int lda,
  __half const *B, int ldb,
  float beta,
  float *C, int ldc) {

  int i = threadIdx.x + blockIdx.x * blockDim.x;
  int j = threadIdx.y + blockIdx.y * blockDim.y;

  if (i < M && j < N) {
    float acc = 0.0f;
    for (int k = 0; k < K; ++k) {
      acc += __half2float(A[i * lda + k]) * __half2float(B[k + j * ldb]);
    }
    C[i + j * ldc] = alpha * acc + beta * C[i + j * ldc];
  }
}

cudaError_t TestBlackwellClusterGemm(int M, int N, int K, float alpha, float beta) {
  int lda = K, ldb = K, ldc = M;

  __half *A, *B;
  float *C_test, *C_ref;
  size_t sizeof_A = sizeof(__half) * M * K;
  size_t sizeof_B = sizeof(__half) * K * N;
  size_t sizeof_C = sizeof(float) * M * N;

  CUDA_CHECK(cudaMalloc(&A, sizeof_A));
  CUDA_CHECK(cudaMalloc(&B, sizeof_B));
  CUDA_CHECK(cudaMalloc(&C_test, sizeof_C));
  CUDA_CHECK(cudaMalloc(&C_ref, sizeof_C));

  {
    int blocks = (M * K + 255) / 256;
    InitHalfMatrix_kernel<<<blocks, 256>>>(A, M * K, 0);
    CUDA_CHECK(cudaGetLastError());
  }
  {
    int blocks = (K * N + 255) / 256;
    InitHalfMatrix_kernel<<<blocks, 256>>>(B, K * N, 17);
    CUDA_CHECK(cudaGetLastError());
  }
  CUDA_CHECK(cudaMemset(C_test, 0, sizeof_C));
  CUDA_CHECK(cudaMemset(C_ref, 0, sizeof_C));

  cudaError_t result = BlackwellClusterGemm(M, N, K, alpha, A, lda, B, ldb, beta, C_test, ldc);
  if (result != cudaSuccess) {
    std::cerr << "Blackwell Cluster GEMM kernel failed" << std::endl;
    cudaFree(A); cudaFree(B); cudaFree(C_test); cudaFree(C_ref);
    return result;
  }

  {
    dim3 block(16, 16);
    dim3 grid((M + 15) / 16, (N + 15) / 16);
    ReferenceGemm_kernel<<<grid, block>>>(M, N, K, alpha, A, lda, B, ldb, beta, C_ref, ldc);
    CUDA_CHECK(cudaGetLastError());
  }
  CUDA_CHECK(cudaDeviceSynchronize());

  // Benchmark (triggered by KH_BENCHMARK=N env var)
  {
    int _kh_trials = kh_benchmark_trials();
    if (_kh_trials > 0 && result == cudaSuccess) {
      kh_benchmark([&]() { BlackwellClusterGemm(M, N, K, alpha, A, lda, B, ldb, beta, C_test, ldc); }, _kh_trials);
    }
  }
  // Benchmark reference kernel
  {
    int _kh_ref_trials = kh_benchmark_trials();
    if (_kh_ref_trials > 0) {
      kh_benchmark([&]() {
        dim3 block(16, 16);
        dim3 grid((M + 15) / 16, (N + 15) / 16);
        ReferenceGemm_kernel<<<grid, block>>>(M, N, K, alpha, A, lda, B, ldb, beta, C_ref, ldc);
      }, _kh_ref_trials, "Ref");
    }
  }



  std::vector<float> h_test(M * N), h_ref(M * N);
  CUDA_CHECK(cudaMemcpy(h_test.data(), C_test, sizeof_C, cudaMemcpyDeviceToHost));
  CUDA_CHECK(cudaMemcpy(h_ref.data(), C_ref, sizeof_C, cudaMemcpyDeviceToHost));

  cudaFree(A); cudaFree(B); cudaFree(C_test); cudaFree(C_ref);

  float max_diff = 0.0f;
  for (int idx = 0; idx < M * N; ++idx) {
    float diff = std::fabs(h_test[idx] - h_ref[idx]);
    max_diff = std::fmax(max_diff, diff);
  }

  float tolerance = 1.0f;
  if (max_diff > tolerance) {
    std::cerr << "Results incorrect. Max diff: " << max_diff << std::endl;
    return cudaErrorUnknown;
  }
  return cudaSuccess;
}

int main(int argc, const char *arg[]) {
  int problem[3] = { 2048, 2048, 2048 };
  for (int i = 1; i < argc && i < 4; ++i) {
    std::stringstream ss(arg[i]);
    ss >> problem[i - 1];
  }

  cudaError_t result = TestBlackwellClusterGemm(problem[0], problem[1], problem[2], 1.0f, 0.0f);
  if (result == cudaSuccess) {
    std::cout << "Passed." << std::endl;
  }
  return result == cudaSuccess ? 0 : -1;
}
